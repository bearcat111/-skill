! =====================================================================
! R1 - 思科(Cisco IOS)出口路由器 : DHCP服务器(全量池) + PAT出公网(ISP1) + OSPF默认路由(metric10优选)
! 互联: G0/0->DSW1(10.0.1.2)  G0/1->DSW2(10.0.2.2)  G0/2->ISP1(100.1.1.1)
! =====================================================================
hostname R1
!
username admin privilege 15 secret Admin@123
!
aaa new-model
aaa authentication login default group radius local
aaa authorization exec default group radius local
aaa accounting exec default start-stop group radius
!
ip domain-name lab.local
crypto key generate rsa modulus 2048
!
! ---------- 环回/管理源 ----------
interface Loopback0
 ip address 10.255.255.1 255.255.255.255
!
! ---------- 内网侧 (NAT inside) ----------
interface GigabitEthernet0/0
 description to DSW1
 ip address 10.0.1.2 255.255.255.252
 ip nat inside
 ip ospf bfd
 bfd interval 100 min_rx 100 multiplier 3
!
interface GigabitEthernet0/1
 description to DSW2
 ip address 10.0.2.2 255.255.255.252
 ip nat inside
 ip ospf bfd
 bfd interval 100 min_rx 100 multiplier 3
!
! ---------- 外网侧 (NAT outside -> ISP1) ----------
interface GigabitEthernet0/2
 description to ISP1
 ip address 100.1.1.1 255.255.255.252
 ip nat outside
!
! ---------- DHCP 排除基础设施地址 (仅分配 .101-.251) ----------
ip dhcp excluded-address 192.168.10.1 192.168.10.100
ip dhcp excluded-address 192.168.10.252 192.168.10.254
ip dhcp excluded-address 192.168.20.1 192.168.20.100
ip dhcp excluded-address 192.168.20.252 192.168.20.254
ip dhcp excluded-address 192.168.30.1 192.168.30.100
ip dhcp excluded-address 192.168.30.252 192.168.30.254
ip dhcp excluded-address 192.168.40.1 192.168.40.100
ip dhcp excluded-address 192.168.40.252 192.168.40.254
ip dhcp excluded-address 192.168.50.1 192.168.50.100
ip dhcp excluded-address 192.168.50.252 192.168.50.254
ip dhcp excluded-address 192.168.60.1 192.168.60.100
ip dhcp excluded-address 192.168.60.252 192.168.60.254
ip dhcp excluded-address 192.168.100.1 192.168.100.100
ip dhcp excluded-address 192.168.100.252 192.168.100.254
!
! ---------- DHCP 全量地址池 (7个VLAN, 仅.101-.254) ----------
ip dhcp pool VLAN10
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.254
!
ip dhcp pool VLAN20
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.254
!
ip dhcp pool VLAN30
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.254
!
ip dhcp pool VLAN40
 network 192.168.40.0 255.255.255.0
 default-router 192.168.40.254
!
ip dhcp pool VLAN50
 network 192.168.50.0 255.255.255.0
 default-router 192.168.50.254
!
ip dhcp pool VLAN60
 network 192.168.60.0 255.255.255.0
 default-router 192.168.60.254
!
ip dhcp pool VLAN100
 network 192.168.100.0 255.255.255.0
 default-router 192.168.100.254
!
! ---------- NAT (ACL 2000 匹配内网, overload复用出口IP) ----------
ip access-list standard NAT-INSIDE
 permit 192.168.0.0 0.0.255.255
!
ip nat inside source list NAT-INSIDE interface GigabitEthernet0/2 overload
!
! ---------- 默认路由 (指向ISP1, 接口down则自动撤销) ----------
ip route 0.0.0.0 0.0.0.0 GigabitEthernet0/2 100.1.1.2
!
! ---------- OSPF area0 (默认优先metric10) ----------
router ospf 1
 router-id 10.255.255.1
 passive-interface default
 no passive-interface GigabitEthernet0/0
 no passive-interface GigabitEthernet0/1
 network 10.0.1.0 0.0.0.3 area 0
 network 10.0.2.0 0.0.0.3 area 0
 network 10.255.255.1 0.0.0.0 area 0
 default-information originate metric 10
!
! ---------- RADIUS (不可达回退本地) ----------
radius server RAD
 address ipv4 192.168.100.100 auth-port 1812 acct-port 1813
 key 0 Radius@123
!
line vty 0 4
 transport input ssh
 login authentication default
!
end
