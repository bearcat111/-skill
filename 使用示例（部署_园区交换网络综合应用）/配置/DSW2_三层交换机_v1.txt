! =====================================================================
! DSW2 - 华为 CE12800 汇聚三层交换机 (主: VLAN20/40/60 ; 备: VLAN10/30/50/100)
! 角色: 各VLAN网关(VRRP) + MST实例2根桥 + OSPF area0 + DSW间LACP三层互联
! =====================================================================
sysname DSW2
!
bfd
!
vlan batch 10 20 30 40 50 60 100
!
! ---------- AAA / SSH ----------
aaa
 authentication-scheme auth_r
  authentication-mode radius local
 authorization-scheme autz_r
  authorization-mode radius local
 accounting-scheme acct_r
  accounting-mode radius local
 domain default
  authentication-scheme auth_r
  authorization-scheme autz_r
  accounting-scheme acct_r
 local-user admin password irreversible-cipher Admin@123
 local-user admin privilege level 15
 local-user admin service-type ssh
!
stelnet server enable
ssh user admin authentication-type password
ssh user admin service-type stelnet
!
interface LoopBack0
 ip address 10.255.255.12 255.255.255.255
!
! ---------- DSW1<->DSW2 LACP ----------
interface Eth-Trunk1
 port link-type trunk
 port trunk allow-pass vlan 10 20 30 40 50 60 100
 mode lacp-static
 load-balance src-dst-ip
!
interface GigabitEthernet1/0/9
 eth-trunk 1
!
interface GigabitEthernet1/0/10
 eth-trunk 1
!
interface Vlanif999
 description DSW2-DSW1 L3 interconnect
 ip address 10.0.0.2 255.255.255.252
 ospf bfd enable
 bfd min-transmit-interval 100 min-receive-interval 100 detect-multiplier 3
!
! ---------- 上联出口路由器 ----------
interface GigabitEthernet1/0/1
 description to R1
 undo portswitch
 ip address 10.0.2.1 255.255.255.252
 ospf bfd enable
 bfd min-transmit-interval 100 min-receive-interval 100 detect-multiplier 3
!
interface GigabitEthernet1/0/3
 description to R2
 undo portswitch
 ip address 10.0.4.1 255.255.255.252
 ospf bfd enable
 bfd min-transmit-interval 100 min-receive-interval 100 detect-multiplier 3
!
! ---------- 下联接入交换机 ----------
interface GigabitEthernet1/0/5
 description to ASW1
 port link-type trunk
 port trunk allow-pass vlan 10 20 100
!
interface GigabitEthernet1/0/6
 description to ASW2
 port link-type trunk
 port trunk allow-pass vlan 30 40 100
!
interface GigabitEthernet1/0/7
 description to ASW3
 port link-type trunk
 port trunk allow-pass vlan 50 60 100
!
! ---------- 业务 VLANIF + VRRP + DHCP中继 ----------
interface Vlanif10
 ip address 192.168.10.253 255.255.255.0
 vrrp vrid 10 virtual-ip 192.168.10.254
 vrrp vrid 10 priority 100
 dhcp select relay
 dhcp relay server-ip 10.0.2.2
 dhcp relay server-ip 10.0.4.2
!
interface Vlanif20
 ip address 192.168.20.253 255.255.255.0
 vrrp vrid 20 virtual-ip 192.168.20.254
 vrrp vrid 20 priority 120
 vrrp vrid 20 preempt-mode timer delay 5
 vrrp vrid 20 track interface GigabitEthernet1/0/1 reduced 30
 dhcp select relay
 dhcp relay server-ip 10.0.4.2
 dhcp relay server-ip 10.0.2.2
!
interface Vlanif30

stp instance 2 root primary

  network 10.0.0.0 0.0.0.3
